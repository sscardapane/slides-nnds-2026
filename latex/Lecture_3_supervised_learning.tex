\section{Supervised learning}
\subsection{Setup and data splits}

\begin{frame}{Supervised learning}

	\begin{tcolorbox}
		A supervised dataset is a set of $N$ input-target pairs:
		\begin{equation}
			\mathcal{S} = \left\{ (x_i, y_i) \right\}_{i=1}^{N} \,.
		\end{equation}
	\end{tcolorbox}

	A model $f_{\boldsymbol{\theta}}$ maps an input $x$ to a prediction $\hat{y}$:
	\begin{equation}
		\hat{y} = f_{\boldsymbol{\theta}}(x) \,.
	\end{equation}

	We fit $\boldsymbol{\theta}$ on $\mathcal{S}$ and evaluate the model on examples that were not used for training.

\end{frame}

\begin{frame}{Training, validation, and test data}

	We split the available examples into three disjoint sets:
	\begin{equation}
		\mathcal{S}
		= {\color{darkred}\mathcal{S}_{\mathrm{train}}}
		\mathbin{\dot\cup} {\color{darkpastelblue}\mathcal{S}_{\mathrm{val}}}
		\mathbin{\dot\cup} {\color{darkgreen}\mathcal{S}_{\mathrm{test}}} \,.
	\end{equation}

	\begin{itemize}
		\item ${\color{darkred}\mathcal{S}_{\mathrm{train}}}$ is used to fit the parameters $\boldsymbol{\theta}$.
		\item ${\color{darkpastelblue}\mathcal{S}_{\mathrm{val}}}$ is used to select the model and its hyper-parameters.
		\item ${\color{darkgreen}\mathcal{S}_{\mathrm{test}}}$ is used for the final evaluation.
	\end{itemize}

	We usually assume independent samples from the same unknown distribution $p(x,y)$ (\textit{identically} and \textit{independently} distributed, i.i.d.). If the deployment distribution differs, we have a \myalert{distribution shift}.

\end{frame}

\begin{frame}{Introducing loss functions}
	
	A \myalert{loss function} assigns a scalar cost to a prediction--target pair:
	\begin{equation}
		\ell\left(\hat{y}, y\right) \in \mathbb{R},
		\qquad \hat{y} = f_{\boldsymbol{\theta}}(x) \,.
	\end{equation}

	A smaller loss means a better prediction. Learning amounts to choosing $\boldsymbol{\theta}$ so that predictions have low loss on the training data.

	\vspace{0.5em}
	For gradient-based learning, we usually choose losses that are \myalert{differentiable} with respect to the model parameters.
	
\end{frame}

\begin{frame}{Expected risk and empirical risk}
	
	Ideally, we would minimize the loss on the entire data distribution:
	\begin{equation}
		\mathcal{R}(\boldsymbol{\theta})
		= \mathbb{E}_{p(x,y)}
		\left[\ell\left(f_{\boldsymbol{\theta}}(x),y\right)\right] \,.
	\end{equation}
	This is the \myalert{expected risk}, but $p(x,y)$ is unknown. We replace the expectation with the average over the training data:
	\begin{equation}
		\mathcal{L}(\boldsymbol{\theta})
		= \frac{1}{N}\sum_{i=1}^{N}
		\ell\left(f_{\boldsymbol{\theta}}(x_i),y_i\right),
		\qquad
		\boldsymbol{\theta}^*=\argmin_{\boldsymbol{\theta}}\mathcal{L}(\boldsymbol{\theta}) \,.
	\end{equation}
	A small empirical risk does not guarantee a small expected risk. Their difference is the \myalert{generalization gap}.
	
\end{frame}

\begin{frame}{Overfitting}
	
	Consider a model that memorizes the training set:
	\begin{equation}
		f(x) =
		\begin{cases}
			{\color{darkred}y_i}
			& \text{if } x=x_i \text{ for some } (x_i,y_i)\in\mathcal{S}_{\mathrm{train}}, \\
			{\color{darkpastelblue}\bar{y}}
			& \text{otherwise}.
		\end{cases}
		\label{eq:lookup_table}
	\end{equation}

	It predicts every training target exactly. For any new input, however, it returns the same default value $\bar{y}$.

	\begin{tcolorbox}
		\begin{equation*}
			{\color{darkred}\mathcal{L}(\boldsymbol{\theta})=0}
			\quad\nRightarrow\quad
			{\color{darkpastelblue}\mathcal{R}(\boldsymbol{\theta})\text{ is small}} \,.
		\end{equation*}
	\end{tcolorbox}

	This is \myalert{overfitting}: fitting the training data without learning a rule that generalizes.
	
\end{frame}

\begin{frame}{Losses for regression}

	\footnotesize
	For \myalert{regression} we have $y,\hat{y}\in\mathbb{R}$, and we can write the prediction error as $e=\hat{y}-y$.

	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.43\textwidth}
			\begin{itemize}
				\item \textcolor{drawred}{\textbf{Squared loss}}:
				${\color{drawred}\ell(e)=e^2}$. Smooth, but gives large errors more weight.

				\item \textcolor{drawgreen}{\textbf{Absolute loss}}:
				${\color{drawgreen}\ell(e)=\lvert e\rvert}$. Less sensitive to outliers, but not differentiable at $e=0$.

				\item \textcolor{drawblue}{\textbf{Huber loss}}:
				quadratic near zero and linear for large $\lvert e\rvert$.
			\end{itemize}
		\end{column}
		\begin{column}{0.54\textwidth}
			\centering
			\includegraphics[width=0.94\linewidth]{images/lecture_3/loss_functions_regression.pdf}
		\end{column}
	\end{columns}

	\vspace{0.15em}
	{\scriptsize Maximum likelihood gives squared loss under Gaussian noise and absolute loss under Laplace noise, see the book.}

\end{frame}

\section{Linear models}
\subsection{Linear models for regression}

\begin{frame}{A linear model}

	\begin{tcolorbox}[boxsep=1mm,top=1mm,bottom=1mm]
		\justifying
		For an input $\vc{x}\in\mathbb{R}^{D}$, a scalar linear model is defined as:
		\begin{equation}
			f_{\boldsymbol{\theta}}(\vc{x})
			= {\color{darkpastelblue}\vc{w}^{\top}\vc{x}}
			  {\color{darkred}+\,b}
			= \sum_{j=1}^{D}w_jx_j+b \,.
		\end{equation}
		The parameters are $\vc{w}\in\mathbb{R}^{D}$ and $b\in\mathbb{R}$, with $\boldsymbol{\theta}=\{\vc{w},b\}$. The scalar $b$ is the \myalert{bias}; we will sometimes omit it to simplify the notation. We call each component $x_j$ a \myalert{feature}, and datasets of this form \myalert{tabular}.
	\end{tcolorbox}

\end{frame}

\begin{frame}{Least-squares cost function}
	
	Combining the squared loss with a linear model results in the \myalert{least-squares} optimization problem:
	
	\begin{equation}
		\text{LS}(\vc{w}) = \frac{1}{n} \sum_{i=1}^n \displaystyle \bigl( y_i - \vc{w}^\top\vc{x}_i \bigr)^2 \,.
	\end{equation}
	%
	We can vectorize LS as:
	%
	\begin{equation}
		\text{LS}(\vc{w}) = \frac{1}{n} \bigl\lVert \stackunder{\vc{y}}{\sss (n)} - \stackunder{\vc{X}}{\sss (n, d)}\stackunder{\vc{w}}{\sss (d)} \bigr\rVert^2 \,,
	\end{equation}
	%
	where $\idx{\vc{X}}{i} = \vc{x}_i$ and $\idx{\vc{y}}{i} = y_i$.
	
\end{frame}

\subsection{Solving the LS problem}

\begin{frame}{Solving the LS problem}
	
	LS is a convex problem, with a simple gradient (\myalert{normal equations}):
	%
	\begin{equation}
		\nabla \text{LS}(\vc{w}) = \frac{2}{n}\vc{X}^\top(\vc{X}\vc{w} - \vc{y}) \,.
	\end{equation}
	%
	LS is special in the sense that $\nabla \text{LS}(\vc{w}) = 0$ is a linear equation that can be solved explicitly:
	%
	\begin{equation}
		\vc{w}^* = \stackunder{\left(\vc{X}^\top\vc{X}\right)}{\sss (d, d)}^{-1}\stackunder{\vc{X}^\top}{\sss (d,n)} \stackunder{\vc{y}}{\sss (n)} = \stackunder{\vc{X}^\top}{\sss (d,n)}\stackunder{\left(\vc{X}\vc{X}^\top\right)}{\sss (n, n)}^{-1} \stackunder{\vc{y}}{\sss (n)} \,.
	\end{equation}
	%
	The terms $\left(\vc{X}^\top\vc{X}\right)^{-1}\vc{X}^\top$ and $\vc{X}^\top\left(\vc{X}\vc{X}^\top\right)^{-1}$ are called the \myalert{pseudoinverses} of $\vc{X}$, and they require the corresponding matrices to be invertible (full rank).
	
\end{frame}

\begin{frame}{Regularizing the LS problem}
	
	Numerical problems in the inversion of $\left(\vc{X}^\top\vc{X}\right)$ can be solved by adding a small amount of $\ell_2$ \myalert{regularization} (\myalert{ridge regression}):
	%
	\begin{equation}
		\text{LS-REG}(\vc{w}) = \text{LS}(\vc{w}) \; {\color{darkred} + \; \frac{\lambda}{2} \lVert \vc{w} \rVert^2} \,,
	\end{equation}
	%
	for some $\lambda > 0$. This makes the problem \textit{strictly} convex and forces the solution to be contained in a ball of given radius, modifying the gradient and the explicit solution as:
	%
	\begin{align}
		\nabla \text{LS-REG}(\vc{w}) & = \nabla \text{LS}(\vc{w}) {\color{darkred} + \lambda\vc{w}} \,. \\
		\vc{w}^* & = \left(\vc{X}^\top\vc{X} {\color{darkred} + \lambda\vc{I}}\right)^{-1}\vc{X}^\top \vc{y} \,.
	\end{align}
	%
	where $\vc{I}$ is the identity matrix of appropriate shape.
	
\end{frame}

\begin{frame}[fragile]{Show me some code!}
	
	Generating some data:

\begin{frame}{Computational complexity of LS}
	
	Matrix multiplication $\shape{\vc{A}}{(a,b)}\shape{\vc{B}}{(b,c)}$ has complexity $\mathcal{O}(abc)$, while matrix inversion has complexity (roughly) cubic.
	
	For large models and datasets, we want algorithms that can scale linearly in both $n$ and $d$: we will see that gradient descent can satisfy this property.
	
\end{frame}

\subsection{Linear models for classification}

\begin{frame}{Multi-class classification}
	
	In \myalert{classification}, $y$ is an integer $\left\{0, \ldots, c-1 \right\}$, such that $y_i = j$ means that $\vc{x}_i$ is of class $j$.
	
	For example, with $c=3$ we might have:
	%
	\begin{itemize}
		\item $y = 0$: the email is spam;
		\item $y = 1$: the email is legit;
		\item $y = 2$: the email is dubious.
	\end{itemize}
	%
	Solving these as regression tasks is generally not an optimal choice: among other things, it is not guaranteed that classes have a definite ordering.
	
\end{frame}

\begin{frame}{Probability distributions}
	
	A common solution is to predict a \myalert{probability distribution} over the classes.
	%
	\begin{tcolorbox}
		A vector $\stackunder{\vc{a}}{\sss (c)}$ belongs to the \myalert{probability simplex} $\Delta_c$ if:
		%
		\begin{equation}
			\sum_i \idx{\vc{a}}{i} = 1 \,, \;\;\; \idx{\vc{a}}{i} \ge 0 \,.
		\end{equation}
	\end{tcolorbox}
	%
	If $f(\vc{x}) = \hat{\vc{y}} \in \Delta_c$, we can interpret it as a \myalert{categorical probability distribution}, e.g., we can select the class with highest probability as:
	%
	\begin{equation}
		\text{class} = \argmax_i \idx{\hat{\vc{y}}}{i} \,.
	\end{equation}
	
\end{frame}

\begin{frame}{A comment on differentiability}
	
	Note that we \textit{cannot} easily predict an integer with our models, because it would require some form of threshold operation which is not compatible with gradient descent (gradient zero almost everywhere).
	
	Predicting a probability distribution can be seen as a \textit{soft} approximation to this problem.
	
\end{frame}

\begin{frame}{The softmax function}
	
	The \myalert{softmax} function maps any vector to the probability simplex:
	%
	\begin{equation}
		\idx{\text{softmax}(\vc{a})}{i} = \frac{\exp\left(a_i\right)}{\sum_j \exp\left(a_j\right)}
	\end{equation}
	%
	The numerator ensures that all outputs are positive, while the denominator ensures that the final vector sums to $1$. It can be seen as a soft approximation to the argmax (a better name is in fact \textit{softargmax}).
	
	If desired, we can control the approximation with an additional hyper-parameter $\tau$ called \myalert{temperature}:
	%
	\begin{equation}
		\idx{\text{softmax}(\vc{a})}{i} = \frac{\exp\left(a_i/\tau\right)}{\sum_j \exp\left(a_j/\tau\right)}
	\end{equation}
	
\end{frame}

\begin{frame}{Final model}
	
	Our linear model for classification becomes:
	%
	\begin{equation}
		\stackunder{f(\vc{x})}{\sss (c)} = \text{softmax}(\stackunder{\vc{W}}{\sss (c,d)} \,\, \stackunder{\vc{x}}{\sss (d)})
	\end{equation}
	%
	The pre-softmax values $\vc{W}\vc{x}$ are called the \myalert{logits} of the model. By explicitly writing the biases $\shape{\vc{b}}{(c)}$:
	%
	\begin{equation}
		f(\vc{x}) = \text{softmax}(\vc{W}\vc{x} + \vc{b}) \,.
	\end{equation}
	
\end{frame}

\begin{frame}{One-hot encoding}
	
	In order to compare the predictions with the ground truth, we encode our targets using a \myalert{one-hot encoding}. Given a pair $(\vc{x}, \vc{y})$:
	%
	\begin{equation}
		y_i = \begin{cases}
			1 & \text{ if } \vc{x} \text{ is of class } i \,, \\
			0 & \text{ otherwise} \,.
		\end{cases}
	\end{equation}
	%
	For example, with $3$ classes $\left\{ \text{cat}, \text{dog}, \text{other} \right\}$:
	%
	\begin{equation}
		\text{cat} = [1, 0, 0] \;\;\; \text{dog} = [0, 1, 0] \;\;\; \text{other} = [0, 0, 1] \,.
	\end{equation}
	%
	This is a probability distribution putting all the \textbf{mass} on a single class.
	
\end{frame}

\begin{frame}{Cross-entropy loss}
	
	Finally, we need a loss function $l$ to compare two probability distributions.
	%
	\begin{tcolorbox}
		The \myalert{cross-entropy} loss is defined for two vectors $\vc{y}, \hat{\vc{y}} \in \Delta_c$ as:
		%
		\begin{equation}
			\text{CE}(\vc{y}, \hat{\vc{y}}) = - \sum_i y_i \log\left(\hat{y}_i\right) \,.
		\end{equation}
	\end{tcolorbox}
	%
	The CE loss can be derived from the maximum likelihood principle under the assumption that $f(\vc{x})$ encodes a \myalert{categorical} distribution (try it!).\footnote[frame]{The equation for the categorical distribution in this case would be $p(\vc{y} \vert f(\vc{x})) = \prod_i \idx{f(\vc{x})}{i}^{y_i}$.}
	
\end{frame}

\begin{frame}{Log-likelihood loss}
	
	The CE loss is generic for \textit{any} pair of vectors belonging to the probability simplex. However, in our case $\vc{y}$ is always a one-hot encoded vector. 
	
	Denote by $t$ the index of the class, $t = \arg\max \vc{y}$, the CE loss simplifies since only one term in the summation will be different from $0$:
	%
	\begin{equation}
		\text{CE}(\vc{y}, \hat{\vc{y}}) = - \log(\hat{y}_t) \,.
	\end{equation}
	%
	Hence, CE can be interpreted as maximizing the probability of the true class at the expense of all other outputs.
	
\end{frame}

\begin{frame}{Logistic regression}
	
	\begin{tcolorbox}
		A \myalert{logistic regression} is a linear model $f(\vc{x}) = \text{softmax}(\vc{W}\vc{x})$ trained by optimizing the cross-entropy:
		%
		\begin{equation}
			\text{LR}(\vc{W}) = \frac{1}{n} \sum_{i=1}^n \text{CE}\left( \vc{y}_i, f(\vc{x}_i) \right) \,.
		\end{equation}
	\end{tcolorbox}
	%
	It is not possible to solve the logistic regression problem explicitly. A linear model for classification has $dc$ parameters (or $dc + c$ with biases).
	
\end{frame}

\begin{frame}{Binary classification}
	
	A special case is \myalert{binary classification}, where $c=2$. In this case, we can predict a single scalar value $f(\vc{x}) \in \left[0, 1\right]$ since:
	%
	\begin{align}
		f(\vc{x}) & \;\;\; \text{ probability of class 1 } \,, \\
		1 - f(\vc{x}) & \;\;\; \text{ probability of class 2 } \,.
	\end{align}
	%
	The softmax function simplifies to the \myalert{sigmoid} function:
	%
	\begin{tcolorbox}
		The sigmoid $\sigma(s) \in \left[0, 1 \right]$ is defined as:
		%
		\begin{equation}
			\sigma(s) = \frac{1}{1 + \exp(-s)} \,.
		\end{equation}
	\end{tcolorbox}
	
\end{frame}

\begin{whiteframe}{Visualizing the sigmoid function}
	
	\begin{figure}
		\centering
		\includegraphics[width=0.6\columnwidth]{images/lecture_3/sigmoid.pdf}
		\vspace{-0.5em}\caption{A visualization of the sigmoid function. Note that $0$ and $1$ are only approached asymptotically.}
	\end{figure}
	
\end{whiteframe}

\begin{frame}{Binary logistic regression model}
	
	Combining everything, we obtain a binary version of the logistic regression algorithm:
	%
	\begin{equation}
		\text{BIN-LR}(\vc{w}) = \frac{1}{n} \sum_{i=1}^n \Bigl[ -\underbrace{y_i\log\left(\sigma(\vc{w}^\top\vc{x})\right)}_{\text{Class 1}} - \underbrace{(1-y_i)\log\left( 1 - \sigma(\vc{w}^\top\vc{x})\right)}_{\text{Class 2}} \Bigr]
	\end{equation}
	%
	In this case, we can obtain the most probable class from the model as:
	%
	\begin{equation}
		\text{class} = \begin{cases}
			1 & \text{ if } \sigma(\vc{w}^\top\vc{x}) > 0.5 \,, \\
			0 & \text{ otherwise } \,.
		\end{cases}
	\end{equation}
	
\end{frame}

\begin{frame}{Gradient of the logistic regression}
	
	By manually differentiating we obtain:
	%
	\begin{equation}
		\sigma'(s) = \sigma(s)(1 - \sigma(s)) \,.
	\end{equation}
	%
	Plugging this into the gradient computation we obtain:
	%
	\begin{equation}
		\nabla \text{BIN-LR}(\vc{w}) = \frac{1}{n} \sum_{i=1}^n (\sigma(\vc{w}^\top\vc{x}_i) - y_i)\vc{x}_i \,,
	\end{equation}
	%
	showing its similarity to the regression case.
	
\end{frame}

\begin{frame}{The logsumexp function}
	
	Why a variant with logits in input? Note that the $i$th term of the cross-entropy wrt the logits $\vc{p}$ is given by:
	%
	\begin{equation}
		- \log\left( \frac{\exp{p_i}}{\sum_j \exp{p_j}} \right) \,.
	\end{equation}
	
	In practice, this can be \textit{highly} unstable. However, it can be rewritten as:
	%
	\begin{equation}
		- p_i + \text{logsumexp}(\vc{p}) \,,
	\end{equation}
	%
	where $\text{logsumexp}$ is defined as $\text{logsumexp}(\vc{p}) = \log\left( \sum_i \exp p_i \right)$.
	
	\blfootnote{\url{https://gregorygundersen.com/blog/2020/02/09/log-sum-exp/}}
	
\end{frame}

\begin{frame}{The logsumexp trick}
	
	The reason this is important is that the logsumexp function is invariant in the following sense:
	%
	\begin{equation}
		\text{logsumexp}(\vc{p}) = \log\left( \sum_i \exp (p_i - c) \right) + c \,,
	\end{equation}
	%
	where $c$ is an arbitrary constant. By setting $c = \max\left(\vc{p}\right)$, we can ensure numerical instabilities never occur.
	
	In this sense, softmax can be interpreted as part of the model \textit{or} as part of the loss; this is not an issue, since $\arg\max$ only cares about the relative ranking of the values, which is not changed by the softmax.
	
\end{frame}

\subsection{Calibration and a probabilistic formulation}

\begin{frame}{Calibration}
	
	A common misconception when doing classification is that $\idx{f(x)}{i}$ can be immediately interpreted as \textit{the probability of pattern $x$ being of class $i$}.
	
	However, this is only true whenever the trained model satisfies:
	%
	\begin{equation}
		p(y = i \,\vert\, x) = \idx{f(x)}{i} \,.
	\end{equation}
	%
	We say the model is well \myalert{calibrated}, but this must be checked manually.
	
	\blfootnote{Guo, C., et al.. \textbf{On calibration of modern neural networks}. ICML 2017.}
	
\end{frame}

\begin{frame}{Measuring calibration}
	
	To measure calibration, we keep a separate validation set, and we split the interval $[0,1]$ into $m$ equispaced bins (each of size $1/m$). Define:
	%
	\begin{itemize}
		\item $B_m$ the number of samples from the validation set, whose predicted confidence falls in bin $m$.
		\item $p_m$ the average confidence of the network for that bin.
		\item $a_m$ the average accuracy of the network for these elements.
	\end{itemize}
	%
	Then, the \myalert{expected calibration error} (ECE) is given by:
	%
	\vspace{1em}
	\begin{equation}
		\displaystyle\text{ECE}= \sum_i \eqnmarkbox[drawred]{node2}{\frac{\lvert\mathcal{B}_i\rvert}{n}} \eqnmarkbox[drawblue]{node}{\left|a_i - p_i\right|}
		\label{eq:ece}
		\end{equation}
		\annotate[yshift=1em]{above,right}{node}{Calibration for bin $i$}
		\annotate[yshift=-1em]{below,right}{node2}{Fraction falling into bin $i$}
	
\end{frame}

\begin{frame}{Focal loss}
	
	This topic is important because more complex networks may be highly over (or under) confident, with many methods to improve it (temperature scaling, logit normalization, ...).
	
	A simple (and popular) option is to decrease the weight given to `easy' samples using a variant of cross-entropy call the \myalert{focal loss}:
	%
	\begin{equation}
		\text{FL}_\alpha(\vc{y}, \hat{\vc{y}}) = -(1 - \hat{\vc{y}}_c)^\alpha \log \hat{\vc{y}}_c \,,
	\end{equation}
	%
	where $c = \arg\max \vc{y}$.
	
	\blfootnote{Mukhoti, J., et al., 2020. \textbf{Calibrating deep neural networks using focal loss}. \textit{Advances in Neural Information Processing Systems}, 33, pp. 15288-15299.}
	
\end{frame}
